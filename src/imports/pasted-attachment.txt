### 4.3 Gönder Butonları

```tsx
{todayTasks.filter(t => t.status === 'draft' || t.status === 'revised').length > 0 && (
  <div className="space-y-3 pt-4">
    <button
      onClick={() => {
        const sendableTasks = todayTasks.filter(t => t.status === 'draft' || t.status === 'revised');
        if (selectedTasks.length === sendableTasks.length) {
          setSelectedTasks([]);
        } else {
          setSelectedTasks(sendableTasks.map(t => t.id));
        }
      }}
      className="w-full flex items-center justify-center gap-2 bg-white/10 hover:bg-white/20 border border-white/20 text-white px-4 py-2.5 rounded-xl font-semibold text-sm transition-all active:scale-95"
    >
      {selectedTasks.length === todayTasks.filter(t => t.status === 'draft' || t.status === 'revised').length ? (
        <>
          <Square className="w-4 h-4" />
          Seçimi Temizle
        </>
      ) : (
        <>
          <CheckSquare className="w-4 h-4" />
          Tümünü Seç ({todayTasks.filter(t => t.status === 'draft' || t.status === 'revised').length})
        </>
      )}
    </button>
    
    <div className="space-y-2">
      {(selectedTasks.length > 0 || selectedOnLeave.length > 0 || selectedStandby.length > 0) && (
        <button
          onClick={() => {
            setSelectedTasks([]);
            setSelectedOnLeave([]);
            setSelectedStandby([]);
```
